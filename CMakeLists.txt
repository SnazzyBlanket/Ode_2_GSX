cmake_minimum_required(VERSION 3.22)
project(OdeToGSX VERSION 0.2.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(JUCE_PATH "" CACHE PATH "Local JUCE source")
set(DAISYSP_PATH "" CACHE PATH "Local DaisySP source")
if(NOT EXISTS "${JUCE_PATH}/CMakeLists.txt" OR NOT EXISTS "${DAISYSP_PATH}/Source/Synthesis/oscillator.cpp")
 message(FATAL_ERROR "Set JUCE_PATH and DAISYSP_PATH to local dependency source directories")
endif()
add_subdirectory("${JUCE_PATH}" "${CMAKE_BINARY_DIR}/juce")
add_library(gsx_core STATIC "${DAISYSP_PATH}/Source/Synthesis/oscillator.cpp")
set_target_properties(gsx_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(gsx_core PUBLIC . "${DAISYSP_PATH}/Source" "${DAISYSP_PATH}/Source/Utility")
set(GSX_FORMATS Standalone VST3)
if(APPLE)
 list(APPEND GSX_FORMATS AU)
endif()
juce_add_plugin(OdeToGSX PRODUCT_NAME "ODE TO GSX" COMPANY_NAME "SnazzyBlanket"
 BUNDLE_ID "com.snazzyblanket.odetogsx" PLUGIN_MANUFACTURER_CODE Snaz PLUGIN_CODE OgSx
 FORMATS ${GSX_FORMATS} IS_SYNTH TRUE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 COPY_PLUGIN_AFTER_BUILD FALSE VST3_AUTO_MANIFEST FALSE)
target_sources(OdeToGSX PRIVATE Plugin.cpp)
target_compile_definitions(OdeToGSX PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(OdeToGSX PRIVATE gsx_core juce::juce_audio_utils)
add_executable(gsx_tests Tests.cpp)
target_link_libraries(gsx_tests PRIVATE gsx_core)
add_executable(plugin_tests PluginTests.cpp)
target_link_libraries(plugin_tests PRIVATE OdeToGSX gsx_core juce::juce_audio_utils)
enable_testing()
add_test(NAME engine COMMAND gsx_tests)
add_test(NAME plugin_state COMMAND plugin_tests)
